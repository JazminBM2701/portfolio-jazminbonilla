Conversión de días (while / do-while)

Programa que convierte un número total de días en años, meses, semanas y días restantes, implementado con bucles while y do while.

Ver código C++ (while) 

#include <iostream>
using namespace std;

int main()
{
    int diasTotales;
    cout << "Ingrese el numero de dias: ";
    cin >> diasTotales;
    int yrs = 0, meses = 0, semanas = 0, dias = 0;

    while (diasTotales >= 365) {
        yrs++;
        diasTotales -= 365;
    }
    while (diasTotales >= 30) {
        meses++;
        diasTotales -= 30;
    }
    while (diasTotales >= 7) {
        semanas++;
        diasTotales -= 7;
    }
    dias = diasTotales;

    cout << yrs << " anios" << endl;
    cout << meses << " meses" << endl;
    cout << semanas << " semanas" << endl;
    cout << dias << " dias" << endl;
    return 0;
}
Ver código C++ (do-while) 

#include <iostream>
using namespace std;

int main()
{
    int diasTotales;
    cout << "Ingrese el numero de dias: ";
    cin >> diasTotales;
    int yrs = 0, meses = 0, semanas = 0, dias = 0;

    if (diasTotales >= 365) {
        do {
            yrs++;
            diasTotales -= 365;
        } while (diasTotales >= 365);
    }
    if (diasTotales >= 30) {
        do {
            meses++;
            diasTotales -= 30;
        } while (diasTotales >= 30);
    }
    if (diasTotales >= 7) {
        do {
            semanas++;
            diasTotales -= 7;
        } while (diasTotales >= 7);
    }
    dias = diasTotales;

    cout << yrs << " anios" << endl;
    cout << meses << " meses" << endl;
    cout << semanas << " semanas" << endl;
    cout << dias << " dias" << endl;
    return 0;
}

Contador con múltiplos (FizzBuzz)

Imprime del 1 al 100, reemplazando múltiplos de 3 y 5 por “fizz”, “buzz” o “fizzbuzz”.

Ver código C++ 

#include <iostream>
using namespace std;

int main(){
    int contador = 1;
    while (contador <= 100 ) {
        if (contador % 3 == 0 && contador % 5 == 0) {
            cout << "fizzbuzz" << endl;
        }
        else if (contador % 3 == 0) {
            cout << "fizz" << endl;
        }
        else if (contador % 5 == 0) {
            cout << "buzz" << endl;
        }
        else {
            cout << contador << endl;
        }
        contador++;
    }
    return 0;
}

Verificador de anagramas

Determina si dos palabras son anagramas (ignora mayúsculas).

Ver código C++ 

#include <iostream>
#include <algorithm>
#include <string>
using namespace std;

bool sonAnagramas(string palabra1, string palabra2) {
    transform(palabra1.begin(), palabra1.end(), palabra1.begin(), ::tolower);
    transform(palabra2.begin(), palabra2.end(), palabra2.begin(), ::tolower);
    if (palabra1 == palabra2) return false;
    sort(palabra1.begin(), palabra1.end());
    sort(palabra2.begin(), palabra2.end());
    return palabra1 == palabra2;
}

int main() {
    string palabra1, palabra2;
    cout << "Ingrese la primera palabra: ";
    cin >> palabra1;
    cout << "Ingrese la segunda palabra: ";
    cin >> palabra2;
    if (sonAnagramas(palabra1, palabra2)) {
        cout << "Las palabras son anagramas." << endl;
    } else {
        cout << "Las palabras NO son anagramas." << endl;
    }
    return 0;
}

Conversor decimal → binario

Convierte un entero decimal a binario sin funciones integradas.

Ver código C++ 

#include <iostream>
#include <string>
using namespace std;

string decimalBinario(int numero) {
    if (numero == 0) return "0";
    string binario = "";
    while (numero > 0) {
        int residuo = numero % 2;
        binario = to_string(residuo) + binario;
        numero /= 2;
    }
    return binario;
}

int main() {
    int numero;
    cout << "Ingrese el numero decimal a convertir : ";
    cin >> numero;
    cout << "El numero en binario es: " << decimalBinario(numero) << endl;
    return 0;
}

Verificador de palíndromos

Limpia el texto (minúsculas, sin tildes, solo alfanumérico) y verifica si es palíndromo.

Ver código C++ 

#include <iostream>
#include <string>
#include <algorithm>
using namespace std;

string limpiarTexto(string texto) {
    string limpio = "";
    for (char c : texto) {
        c = tolower(c);
        if (c == 'á') c = 'a';
        else if (c == 'é') c = 'e';
        else if (c == 'í') c = 'i';
        else if (c == 'ó') c = 'o';
        else if (c == 'ú') c = 'u';
        if (isalnum(c)) limpio += c;
    }
    return limpio;
}

bool esPalindromo(string texto) {
    string limpio = limpiarTexto(texto);
    string invertido = limpio;
    reverse(invertido.begin(), invertido.end());
    return limpio == invertido;
}

int main() {
    string texto;
    cout << "Ingrese un texto: ";
    getline(cin, texto); 
    if (esPalindromo(texto)) {
        cout << "El texto es un palindromo." << endl;
    } else {
        cout << "El texto NO es un palindromo." << endl;
    }
    return 0;
}

MCD y MCM

Implementación del algoritmo de Euclides (iterativo) para MCD y fórmula del MCM.

Ver código C++ 

#include <iostream>
#include <cstdlib>
using namespace std;

int MCD(int a, int b) {
    while (b != 0) {
        int residuo = a % b;
        a = b;
        b = residuo;
    }
    return abs(a);
}

int mcm(int a, int b) {
    return abs(a * b) / MCD(a, b);
}

int main() {
    int num1, num2;
    cout << "Ingrese el primer numero: ";
    cin >> num1;
    cout << "Ingrese el segundo numero: ";
    cin >> num2;
    cout << "El MCD de " << num1 << " y " << num2 << " es: " << MCD(num1, num2) << endl;
    cout << "El mcm de " << num1 << " y " << num2 << " es: " << mcm(num1, num2) << endl;
    return 0;
}
